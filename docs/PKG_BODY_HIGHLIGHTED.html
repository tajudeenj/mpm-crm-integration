14 /* ====================================================================
15 GENERATE_UNIQUE_ID
16 Builds x-unique-id for the APIC header: <channel_id><timestamp><seq>
17 Format driven entirely by CRM_MPM_UNIQUE_ID_CONFIG (single row) so
18 the standard can change later without touching this package.
19 Sequence portion comes from CRM_MPM_UNIQUE_ID_SEQ (DB sequence,
20 CYCLE'd automatically — no manual reset needed, concurrency-safe
21 across multiple sessions since Oracle sequences are atomic).
22 Mirrors Java's UniqueIdGenerator exactly — same channel/format/width
23 by default (814 / YYYYMMDDHH24MISSFF3 / 4 digits).
24 ==================================================================== */
25 FUNCTION GENERATE_UNIQUE_ID RETURN VARCHAR2 IS
26 v_channel_id VARCHAR2(10);
27 v_timestamp_format VARCHAR2(50);
28 v_sequence_digits NUMBER;
29 v_timestamp VARCHAR2(50);
30 v_seq NUMBER;
31 v_seq_padded VARCHAR2(20);
32 BEGIN
33 SELECT CHANNEL_ID, TIMESTAMP_FORMAT, SEQUENCE_DIGITS
34 INTO v_channel_id, v_timestamp_format, v_sequence_digits
35 FROM CRM_MPM_UNIQUE_ID_CONFIG
36 WHERE CONFIG_ID = 1
37 AND IS_ACTIVE = 'Y';
38
39 v_timestamp := TO_CHAR(SYSTIMESTAMP, v_timestamp_format);
40
41 SELECT CRM_MPM_UNIQUE_ID_SEQ.NEXTVAL INTO v_seq FROM DUAL;
42
43 v_seq_padded := LPAD(TO_CHAR(v_seq), v_sequence_digits, '0');
44
45 RETURN v_channel_id || v_timestamp || v_seq_padded;
46
47 EXCEPTION
48 WHEN NO_DATA_FOUND THEN
49 -- Config missing or inactive — fall back to a safe default
50 -- so SEND_TO_APIC never fails purely because of this header.
51 RETURN '814' || TO_CHAR(SYSTIMESTAMP, 'YYYYMMDDHH24MISSFF3') ||
52 LPAD(TO_CHAR(CRM_MPM_UNIQUE_ID_SEQ.NEXTVAL), 4, '0');
53 WHEN OTHERS THEN
54 -- Never let unique-id generation fail the whole send — log and
55 -- fall back to a timestamp-only id so the request still goes out.
56 RETURN '814' || TO_CHAR(SYSTIMESTAMP, 'YYYYMMDDHH24MISSFF3');
57 END GENERATE_UNIQUE_ID;

541 BEGIN UTL_HTTP.END_RESPONSE(v_http_resp); EXCEPTION WHEN OTHERS THEN NULL; END;
542 v_status_code := NULL;
543 v_final_status := 'FAILED';
544 v_error_code := 'APIC_ERROR';
545 v_error_message := 'HTTP client/server error: ' || v_err_msg;
546 WHEN OTHERS THEN
547 v_err_msg := SQLERRM;
548 BEGIN UTL_HTTP.END_RESPONSE(v_http_resp); EXCEPTION WHEN OTHERS THEN NULL; END;
549 v_status_code := NULL;
550 v_final_status := 'FAILED';
551 v_error_code := 'UNKNOWN_ERROR';
552 v_error_message := 'Unexpected error: ' || v_err_msg;
553 END;
554
555 UPDATE CRM_MPM_CRM_INTEGRATION_LOG
556 SET HTTP_STATUS_CODE = v_status_code,
557 ACK_RESPONSE = v_response,
558 ACK_REQUEST_ID = v_ack_request_id, -- request_id from CRM ACK
559 ACK_STATUS_CODE = v_ack_status_code, -- '0000' or '9999'
560 ACK_DESCRIPTION = v_ack_description,
561 ACK_RESPONSE_TIMESTAMP = v_ack_response_ts, -- response_timestamp from LEG 1
562 FINAL_STATUS = v_final_status,
563 ERROR_CODE = v_error_code,
564 ERROR_MESSAGE = v_error_message,
565 X_UNIQUE_ID = v_unique_id,
566 RETRY_COUNT = p_attempt_no - 1,
567 UPDATED_DATE = SYSTIMESTAMP
568 WHERE LOG_ID = v_log_id;
569
570 COMMIT;
571
572 EXCEPTION
573 WHEN OTHERS THEN
574 -- FIX: capture SQLERRM before embedding in UPDATE
575 v_err_msg := SQLERRM;
576 IF v_log_id IS NOT NULL THEN
577 UPDATE CRM_MPM_CRM_INTEGRATION_LOG
578 SET FINAL_STATUS = 'FAILED',
579 ERROR_CODE = 'UNKNOWN_ERROR',
580 ERROR_MESSAGE = 'SEND_TO_APIC fatal error: ' || v_err_msg,
581 UPDATED_DATE = SYSTIMESTAMP
582 WHERE LOG_ID = v_log_id;
583 COMMIT;
584 END IF;
585 RAISE;
586 END SEND_TO_APIC;

589 /* ====================================================================
590 RUN_OUTBOUND_JOB
591 ==================================================================== */
592 PROCEDURE RUN_OUTBOUND_JOB IS
593 v_run_id NUMBER;
594 v_processed NUMBER := 0;
595 v_success NUMBER := 0;
596 v_failed NUMBER := 0;
597 v_max_ts TIMESTAMP;
598 v_log_id NUMBER;
599 v_cursor_id INTEGER;
600 v_sql VARCHAR2(4000);
601 v_key_val VARCHAR2(4000);
602 v_filter_val VARCHAR2(4000);
603 v_status VARCHAR2(20);
604 -- FIX: local var to capture SQLERRM before DML
605 v_err_msg VARCHAR2(4000);
606 BEGIN
607 -- ----------------------------------------------------------------
608 -- JOB-LOCK GUARD: prevent concurrent overlapping executions.
609 -- If another OUTBOUND_PUSH has been RUNNING for more than 5 minutes
610 -- (i.e. it is a real overlap, not the current insert we are about to do)
611 -- skip this invocation entirely.
612 -- ----------------------------------------------------------------
613 DECLARE
614 v_running_count NUMBER;
615 BEGIN
616 SELECT COUNT(*) INTO v_running_count
617 FROM CRM_MPM_JOB_RUN_HISTORY
618 WHERE JOB_NAME = 'OUTBOUND_PUSH'
619 AND STATUS = 'RUNNING'
620 AND START_TIME < SYSTIMESTAMP - (5/1440); -- older than 5 min = real overlap
621
622 IF v_running_count > 0 THEN
623 RETURN; -- previous run still active, skip this invocation
624 END IF;
625 END;
626
627 INSERT INTO CRM_MPM_JOB_RUN_HISTORY (JOB_NAME, STATUS)
628 VALUES ('OUTBOUND_PUSH', 'RUNNING')
629 RETURNING RUN_ID INTO v_run_id;
630 COMMIT;
631
632 FOR reg IN (
633 SELECT r.*, w.LAST_PROCESSED_TS
634 FROM CRM_MPM_API_REGISTRY r
635 JOIN CRM_MPM_API_WATERMARK w ON w.REGISTRY_ID = r.REGISTRY_ID
636 WHERE r.IS_ACTIVE = 'Y'
637 ) LOOP
638
639 v_max_ts := reg.LAST_PROCESSED_TS;
640
641 v_sql := 'SELECT ' || reg.SOURCE_KEY_COL || ', ' || reg.SOURCE_FILTER_COL ||
642 ' FROM ' || reg.SOURCE_VIEW ||
643 ' WHERE ' || reg.SOURCE_FILTER_COL || ' > :wm' ||
644 ' ORDER BY ' || reg.SOURCE_FILTER_COL;
645
646 v_cursor_id := DBMS_SQL.OPEN_CURSOR;
647 DBMS_SQL.PARSE(v_cursor_id, v_sql, DBMS_SQL.NATIVE);
648 DBMS_SQL.BIND_VARIABLE(v_cursor_id, ':wm', reg.LAST_PROCESSED_TS);
649 DBMS_SQL.DEFINE_COLUMN(v_cursor_id, 1, v_key_val, 4000);
650 DBMS_SQL.DEFINE_COLUMN(v_cursor_id, 2, v_filter_val, 4000);
651 -- FIX: DBMS_SQL.EXECUTE is a FUNCTION returning INTEGER; must capture return value
652 DECLARE v_exec_rows INTEGER; BEGIN v_exec_rows := DBMS_SQL.EXECUTE(v_cursor_id); END;
653
654 LOOP
655 EXIT WHEN DBMS_SQL.FETCH_ROWS(v_cursor_id) = 0;
656
657 DBMS_SQL.COLUMN_VALUE(v_cursor_id, 1, v_key_val);
658 DBMS_SQL.COLUMN_VALUE(v_cursor_id, 2, v_filter_val);
659
660 v_processed := v_processed + 1;
661
662 BEGIN
663 SEND_TO_APIC(
664 p_registry_id => reg.REGISTRY_ID,
665 p_key_value => v_key_val,
666 p_attempt_no => 1,
667 p_log_id_out => v_log_id
668 );
669
670 -- FIX: use separate variable v_status instead of reusing v_filter_val
671 SELECT FINAL_STATUS INTO v_status
672 FROM CRM_MPM_CRM_INTEGRATION_LOG
673 WHERE LOG_ID = v_log_id;
674
675 IF v_status = 'SENT' THEN
676 v_success := v_success + 1;
677 -- ------------------------------------------------
678 -- WATERMARK ADVANCE PER ROW (Fix 3 — crash safety)
679 -- Advance immediately after each successful send so
680 -- a mid-run crash never re-sends already-sent rows.
681 -- v_filter_val holds the SOURCE_FILTER_COL value for
682 -- this row (fetched as VARCHAR2 via DBMS_SQL above).
683 -- ------------------------------------------------
684 BEGIN
685 UPDATE CRM_MPM_API_WATERMARK
686 SET LAST_PROCESSED_TS = TO_TIMESTAMP(v_filter_val,
687 'YYYY-MM-DD HH24:MI:SS.FF6'),
688 UPDATED_DATE = SYSTIMESTAMP
689 WHERE REGISTRY_ID = reg.REGISTRY_ID
690 AND (LAST_PROCESSED_TS IS NULL
691 OR LAST_PROCESSED_TS < TO_TIMESTAMP(v_filter_val,
692 'YYYY-MM-DD HH24:MI:SS.FF6'));
693 EXCEPTION
694 WHEN OTHERS THEN NULL; -- non-fatal; end-of-loop update covers it
695 END;
696 ELSE
697 v_failed := v_failed + 1;
698 END IF;
699
700 EXCEPTION
701 WHEN OTHERS THEN
702 v_failed := v_failed + 1;
703 END;
704 END LOOP;
705
706 DBMS_SQL.CLOSE_CURSOR(v_cursor_id);
707
708 -- Final watermark update (covers edge case where filter_col is
709 -- not a parseable TIMESTAMP string — falls back to MAX query).
710 BEGIN
711 EXECUTE IMMEDIATE
712 'SELECT MAX(' || reg.SOURCE_FILTER_COL || ') FROM ' || reg.SOURCE_VIEW ||
713 ' WHERE ' || reg.SOURCE_FILTER_COL || ' > :wm'
714 INTO v_max_ts
715 USING reg.LAST_PROCESSED_TS;
716
717 IF v_max_ts IS NOT NULL THEN
718 UPDATE CRM_MPM_API_WATERMARK
719 SET LAST_PROCESSED_TS = v_max_ts,
720 LAST_RUN_STATUS = 'COMPLETED',
721 LAST_RUN_RECORDS = v_processed,
722 UPDATED_DATE = SYSTIMESTAMP
723 WHERE REGISTRY_ID = reg.REGISTRY_ID;
724 END IF;
725 EXCEPTION
726 WHEN OTHERS THEN
727 v_err_msg := SQLERRM;
728 UPDATE CRM_MPM_API_WATERMARK
729 SET LAST_RUN_STATUS = 'ERROR: ' || SUBSTR(v_err_msg, 1, 200)
730 WHERE REGISTRY_ID = reg.REGISTRY_ID;
731 END;
732
733 COMMIT;
734 END LOOP;
735
736 UPDATE CRM_MPM_JOB_RUN_HISTORY
737 SET END_TIME = SYSTIMESTAMP,
738 RECORDS_PROCESSED = v_processed,
739 RECORDS_SUCCESS = v_success,
740 RECORDS_FAILED = v_failed,
741 STATUS = 'COMPLETED'
742 WHERE RUN_ID = v_run_id;
743 COMMIT;
744
745 EXCEPTION
746 WHEN OTHERS THEN
747 v_err_msg := SQLERRM;
748 IF DBMS_SQL.IS_OPEN(v_cursor_id) THEN
749 DBMS_SQL.CLOSE_CURSOR(v_cursor_id);
750 END IF;
751 UPDATE CRM_MPM_JOB_RUN_HISTORY
752 SET END_TIME = SYSTIMESTAMP,
753 STATUS = 'ERROR',
754 ERROR_MESSAGE = SUBSTR(v_err_msg, 1, 4000)
755 WHERE RUN_ID = v_run_id;
756 COMMIT;
757 RAISE;
758 END RUN_OUTBOUND_JOB;

827 -- ── Read callback status code from master table ────────────────
828 FUNCTION GET_CB_CODE(p_code IN VARCHAR2) RETURN VARCHAR2 IS
829 v_code VARCHAR2(20);
830 BEGIN
831 SELECT ERROR_CODE INTO v_code
832 FROM CRM_MPM_ERROR_CODE_MASTER
833 WHERE ERROR_CODE = p_code;
834 RETURN v_code;
835 EXCEPTION
836 WHEN NO_DATA_FOUND THEN RETURN p_code;
837 WHEN OTHERS THEN RETURN p_code;
838 END GET_CB_CODE;
839
840 BEGIN
841 -- ── LEVEL 1: Log receipt immediately ─────────────────────────────
842 WRITE_AUDIT('RECEIVED',
843 'Callback received from ESB. service=' || NVL(p_service_name,'(null)') ||
844 ' x-unique-id=' || NVL(p_x_unique_id,'(null)') ||
845 ' channel=' || NVL(p_channel_id,'(null)') ||
846 ' payload_len=' || NVL(TO_CHAR(DBMS_LOB.GETLENGTH(p_callback_payload)),'0'));
847
848 -- ── LEVEL 2: Validate service name ───────────────────────────────
849 BEGIN
850 SELECT * INTO v_reg
851 FROM CRM_MPM_API_REGISTRY
852 WHERE SERVICE_NAME = p_service_name AND IS_ACTIVE = 'Y';
853 EXCEPTION
854 WHEN NO_DATA_FOUND THEN
855 WRITE_AUDIT('FAILED',
856 'Unknown SERVICE_NAME [' || p_service_name || '] — not found in CRM_MPM_API_REGISTRY');
857 p_status_code := GET_CB_CODE('CB_1001');
858 p_description := 'Unknown SERVICE_NAME [' || p_service_name || ']';
859 p_result_out :=
860 '{"status":"ERROR"' ||
861 ',"service_type":"' || NVL(p_service_name,'') || '"' ||
862 ',"message":"Unknown SERVICE_NAME [' || p_service_name || ']"' ||
863 '}';
864 WRITE_AUDIT('FAILED', p_description, NULL,
865 p_result_out, p_status_code, p_description);
866 RETURN;
867 END;
868
869 WRITE_AUDIT('VALIDATING',
870 'Registry found. REGISTRY_ID=' || v_reg.REGISTRY_ID ||
871 ' Parsing callback JSON...');
872
873 -- Parse callback fields — confirmed JSON contract (6 cases spec):
874 -- { "request_id":"...", "status":"0000"/"9999",
875 -- "description":"...", "record_type":"...", "record_action":"...",
876 -- "message":"...", "callback_timestamp":"...",
877 -- "crm_entity_id":"D365-GUID or null",
878 -- "processing_result": { "result_code":"SUCCESS/VALIDATION_FAILED/
879 -- RECORD_NOT_FOUND/DUPLICATE_RECORD", "result_description":"...",
880 -- "validation_errors":[{"field":"","error_code":"","error_message":""}] } }
881 -- Spec confirmed: field name is "request_id" (with underscore).
882 -- "requestid" (no underscore) was only used during manual Postman testing.
883 -- Try spec field name first, fall back to real APIC field name
884 v_crm_reference := JSON_VALUE(p_callback_payload, '$.request_id');
885 IF v_crm_reference IS NULL THEN
886 v_crm_reference := JSON_VALUE(p_callback_payload, '$.requestid');
887 END IF;
888 v_callback_message := JSON_VALUE(p_callback_payload, '$.message');
889
890 WRITE_AUDIT('MATCHING',
891 'Parsed request_id=' || NVL(v_crm_reference,'(null)') ||
892 ' Searching CRM_MPM_CRM_INTEGRATION_LOG by ACK_REQUEST_ID...');
893
894 -- Primary lookup: by ACK_REQUEST_ID = request_id from callback JSON
895 BEGIN
896 SELECT LOG_ID, FINAL_STATUS INTO v_log_id, v_current_status
897 FROM CRM_MPM_CRM_INTEGRATION_LOG
898 WHERE REGISTRY_ID = v_reg.REGISTRY_ID
899 AND ACK_REQUEST_ID = v_crm_reference
900 ORDER BY LOG_ID DESC
901 FETCH FIRST 1 ROW ONLY;
902
903 WRITE_AUDIT('MATCHED',
904 'Log row found by ACK_REQUEST_ID. LOG_ID=' || v_log_id ||
905 ' current FINAL_STATUS=' || v_current_status, v_log_id);
906 EXCEPTION
907 WHEN NO_DATA_FOUND THEN
908 -- request_id not found in ACK_REQUEST_ID column.
909 -- NO FALLBACK — we never guess which log row to update.
910 -- If request_id is wrong or missing, reject the callback cleanly.
911 -- ESB must send the exact request_id returned in LEG 1 ACK.
912 WRITE_AUDIT('NO_LOG_MATCH',
913 'No log row found for ACK_REQUEST_ID=' ||
914 NVL(v_crm_reference,'(null)') ||
915 ' SERVICE_NAME=' || p_service_name ||
916 ' — rejected. ESB must send exact request_id from LEG 1 ACK.');
917 p_status_code := GET_CB_CODE('CB_1002');
918 p_description := 'No matching log entry. request_id=' ||
919 NVL(v_crm_reference,'(null)') ||
920 ' not found for SERVICE_NAME=' || p_service_name;
921 p_result_out :=
922 '{"status":"ERROR"' ||
923 ',"service_type":"' || NVL(p_service_name,'') || '"' ||
924 ',"message":"No matching log entry. request_id=' ||
925 NVL(v_crm_reference,'(null)') ||
926 ' not found in ACK_REQUEST_ID for SERVICE_NAME=' ||
927 p_service_name || '. Use exact request_id from LEG 1 ACK response."' ||
928 '}';
929 WRITE_AUDIT('NO_LOG_MATCH',
930 'Rejected — response sent to ESB',
931 NULL, p_result_out, p_status_code, p_description);
932 RETURN;
933 END;
934
935 -- Parse callback JSON body to get result_code and crm_entity_id
936 DECLARE
937 v_result_code VARCHAR2(50);
938 v_result_desc VARCHAR2(500);
939 v_crm_ent_id VARCHAR2(200);
940 v_valid_errors CLOB;
941 v_cb_status VARCHAR2(10);
942 BEGIN
943 -- Parse all LEG 2 callback fields per confirmed 6-case spec
944 v_cb_status := JSON_VALUE(p_callback_payload, '$.status');
945 v_crm_ent_id := JSON_VALUE(p_callback_payload, '$.crm_entity_id');
946 v_result_code := JSON_VALUE(p_callback_payload, '$.processing_result.result_code');
947 v_result_desc := JSON_VALUE(p_callback_payload, '$.processing_result.result_description');
948
949 -- Extract validation_errors array as JSON string for storage
950 BEGIN
951 SELECT JSON_QUERY(p_callback_payload, '$.processing_result.validation_errors')
952 INTO v_valid_errors FROM DUAL;
953 EXCEPTION WHEN OTHERS THEN v_valid_errors := NULL;
954 END;
955
956 -- Map processing_result.result_code → FINAL_STATUS
957 -- This is the authoritative field — top-level status (0000/9999) in LEG 2
958 -- only indicates whether D365 processing succeeded or failed broadly.
959 -- The specific reason is always in result_code.
960 -- CASE 1: result_code=SUCCESS → FINAL_STATUS=SUCCESS
961 -- CASE 2: result_code=VALIDATION_FAILED → FINAL_STATUS=VALIDATION_FAILED
962 -- CASE 3: result_code=RECORD_NOT_FOUND → FINAL_STATUS=RECORD_NOT_FOUND
963 -- CASE 4: result_code=DUPLICATE_RECORD → FINAL_STATUS=DUPLICATE_RECORD
964 -- CASE 5: result_code=INVALID_JSON → FINAL_STATUS=VALIDATION_FAILED
965 -- CASE 6: result_code=CONFIG_NOT_FOUND → FINAL_STATUS=CRM_REJECTED
966 -- CASE 7: result_code=MANDATORY_FIELD_MISSING→ FINAL_STATUS=VALIDATION_FAILED
967 -- CASE 8: result_code=PRIMARY_FIELD_MISSING → FINAL_STATUS=VALIDATION_FAILED
968 -- CASE 9: result_code=LOOKUP_VALIDATION_FAILED→ FINAL_STATUS=RECORD_NOT_FOUND
969 -- CASE 10:result_code=FAILURE → FINAL_STATUS=FAILED
970 IF v_result_code = 'SUCCESS' OR v_cb_status = '0000' THEN
971 v_final_status := 'SUCCESS';
972 v_error_code := 'SUCCESS';
973 ELSIF v_cb_status = '9999' AND v_result_code IS NULL THEN
974 -- status=9999 with no result_code — CRM failed, record not created
975 -- Confirmed by CRM team: 9999 = failure, must retry from Oracle
976 v_final_status := 'FAILED';
977 v_error_code := 'FAILURE';
978 ELSIF v_result_code = 'VALIDATION_FAILED' THEN
979 v_final_status := 'VALIDATION_FAILED';
980 v_error_code := 'VALIDATION_FAILED';
981 ELSIF v_result_code = 'RECORD_NOT_FOUND' THEN
982 v_final_status := 'RECORD_NOT_FOUND';
983 v_error_code := 'RECORD_NOT_FOUND';
984 ELSIF v_result_code = 'DUPLICATE_RECORD' THEN
985 v_final_status := 'DUPLICATE_RECORD';
986 v_error_code := 'DUPLICATE_RECORD';
987 -- New CRM error codes confirmed by CRM team
988 ELSIF v_result_code = 'INVALID_JSON' THEN
989 -- Bad JSON payload -- data fix needed, no retry
990 v_final_status := 'VALIDATION_FAILED';
991 v_error_code := 'INVALID_JSON';
992 ELSIF v_result_code = 'CONFIG_NOT_FOUND' THEN
993 -- PxRM integration master missing -- config fix needed
994 v_final_status := 'CRM_REJECTED';
995 v_error_code := 'CONFIG_NOT_FOUND';
996 ELSIF v_result_code = 'MANDATORY_FIELD_MISSING' THEN
997 -- Required field absent in payload -- data fix needed
998 v_final_status := 'VALIDATION_FAILED';
999 v_error_code := 'MANDATORY_FIELD_MISSING';
1000 ELSIF v_result_code = 'PRIMARY_FIELD_MISSING' THEN
1001 -- Oracle primary key missing in payload -- mapping fix needed
1002 v_final_status := 'VALIDATION_FAILED';
1003 v_error_code := 'PRIMARY_FIELD_MISSING';
1004 ELSIF v_result_code = 'LOOKUP_VALIDATION_FAILED' THEN
1005 -- Parent/related record not found in PxRM -- retryable after parent pushed
1006 v_final_status := 'RECORD_NOT_FOUND';
1007 v_error_code := 'LOOKUP_VALIDATION_FAILED';
1008 ELSIF v_result_code = 'FAILURE' THEN
1009 -- Generic PostOperation catch -- retryable
1010 v_final_status := 'FAILED';
1011 v_error_code := 'FAILURE';
1012 ELSE
1013 -- Unknown result_code — validate against master table
1014 v_final_status := 'FAILED';
1015 BEGIN
1016 SELECT ERROR_CODE INTO v_error_code
1017 FROM CRM_MPM_ERROR_CODE_MASTER
1018 WHERE ERROR_CODE = NVL(v_result_code, 'UNKNOWN_ERROR');
1019 EXCEPTION
1020 WHEN NO_DATA_FOUND THEN v_error_code := 'UNKNOWN_ERROR';
1021 END;
1022 END IF;
1023
1024 -- ----------------------------------------------------------------
1025 -- 9999 OVERRIDE: CRM team confirmed status=9999 means record was
1026 -- NOT created in PxRM regardless of result_code.
1027 -- Therefore ALL 9999 responses must be retried from Oracle EXCEPT
1028 -- DUPLICATE_RECORD (record already exists — retry is pointless).
1029 -- This override runs AFTER result_code mapping so error_code
1030 -- still captures the specific reason for the failure.
1031 -- ----------------------------------------------------------------
1032 IF v_cb_status = '9999'
1033 AND NVL(v_result_code,'x') != 'DUPLICATE_RECORD'
1034 AND NVL(v_final_status,'x') != 'SUCCESS' THEN
1035 v_final_status := 'FAILED';
1036 -- Keep v_error_code as-is so specific reason is preserved
1037 -- e.g. MANDATORY_FIELD_MISSING stays as error code
1038 -- but FINAL_STATUS=FAILED means retry job picks it up
1039 END IF;
1040
1041 -- Store crm_entity_id and validation_errors via additional UPDATE
1042 UPDATE CRM_MPM_CRM_INTEGRATION_LOG
1043 SET CRM_ENTITY_ID = v_crm_ent_id,
1044 CALLBACK_RESULT_CODE = v_result_code,
1045 CALLBACK_RESULT_DESC = v_result_desc,
1046 CALLBACK_VALID_ERRORS = v_valid_errors
1047 WHERE LOG_ID = v_log_id;
1048 END;
1049
1050 UPDATE CRM_MPM_CRM_INTEGRATION_LOG
1051 SET CRM_REFERENCE_NO = NVL(CRM_REFERENCE_NO, v_crm_reference),
1052 FINAL_STATUS = v_final_status,
1053 ERROR_CODE = v_error_code,
1054 ERROR_MESSAGE = v_callback_message,
1055 CALLBACK_PAYLOAD = p_callback_payload,
1056 CALLBACK_DATE = SYSTIMESTAMP,
1057 CALLBACK_X_UNIQUE_ID = p_x_unique_id,
1058 CALLBACK_CHANNEL_ID = p_channel_id,
1059 IS_FINAL_ATTEMPT = CASE WHEN v_final_status IN ('SUCCESS','ALREADY_PROCESSED')
1060 THEN 'Y' ELSE IS_FINAL_ATTEMPT END,
1061 UPDATED_DATE = SYSTIMESTAMP
1062 WHERE LOG_ID = v_log_id;
1063
1064 DECLARE
1065 v_source_rec_id VARCHAR2(4000);
1066 BEGIN
1067 SELECT SOURCE_RECORD_ID INTO v_source_rec_id
1068 FROM CRM_MPM_CRM_INTEGRATION_LOG
1069 WHERE LOG_ID = v_log_id;
1070
1071 IF v_reg.POST_CALLBACK_PROC IS NOT NULL THEN
1072 -- Custom post-callback procedure defined — call it
1073 v_sql := 'BEGIN ' || v_reg.POST_CALLBACK_PROC ||
1074 '(:p_key, :p_status, :p_ref, :p_errcode, :p_errmsg); END;';
1075 EXECUTE IMMEDIATE v_sql
1076 USING v_source_rec_id, v_final_status, v_crm_reference,
1077 v_error_code, v_callback_message;
1078
1079 ELSIF v_reg.CALLBACK_TARGET_TABLE IS NOT NULL
1080 AND v_reg.CALLBACK_STATUS_COL IS NOT NULL
1081 AND v_reg.CALLBACK_REF_COL IS NOT NULL
1082 AND v_reg.CALLBACK_KEY_COL IS NOT NULL
1083 THEN
1084 -- Update entity table with CRM sync status — only if table is configured
1085 -- and actually exists. Skip gracefully if table does not exist yet.

1089 v_reg.CALLBACK_REF_COL || ' = :p_ref ' ||
1090 ' WHERE ' || v_reg.CALLBACK_KEY_COL || ' = :p_key';

1092 USING v_final_status, v_crm_reference, v_source_rec_id;

1095 -- Table does not exist or column missing — log and continue.
1096 -- The integration log (CRM_MPM_CRM_INTEGRATION_LOG) is already

1099 END;
1100 END IF;
1101 -- If no callback table configured at all — skip silently.
1102 -- CRM_MPM_CRM_INTEGRATION_LOG already has the full result.
1103 END;
1104
1105 COMMIT;

1111 SELECT NVL(CRM_ENTITY_ID,'')

1120 ',"final_status":"' || v_final_status || '"' ||

1129 ',"service_type":"' || p_service_name || '"' ||
1130 ',"log_id":' || v_log_id ||

1166 /* ====================================================================
1167 RUN_TIMEOUT_JOB
1168 ==================================================================== */
1169 PROCEDURE RUN_TIMEOUT_JOB IS
1170 -- FIX: Declare an explicit named cursor — implicit FOR loop cursors over
1171 -- views can raise PLS-00364 and ORA-00904 when their columns are
1172 -- referenced in DML or assignments inside the loop body.
1173 -- Query joins LOG directly to REGISTRY to get MAX_RETRY_COUNT.
1174 CURSOR c_timeout IS
1175 SELECT t.LOG_ID,
1176 t.REGISTRY_ID,
1177 t.RETRY_COUNT,
1178 r.MAX_RETRY_COUNT,
1179 r.RETRY_INTERVAL_MINUTES
1180 FROM CRM_MPM_CRM_INTEGRATION_LOG t
1181 JOIN CRM_MPM_API_REGISTRY r ON r.REGISTRY_ID = t.REGISTRY_ID
1182 WHERE t.FINAL_STATUS IN ('SENT','PENDING')
1183 AND t.SENT_DATE < SYSTIMESTAMP - (r.TIMEOUT_MINUTES / 1440)
1184 AND r.IS_ACTIVE = 'Y';
1185
1186 v_run_id NUMBER;
1187 v_count NUMBER := 0;
1188 -- Scalar locals populated from each fetched row
1189 v_log_id NUMBER;
1190 v_retry_count NUMBER;
1191 v_max_retry NUMBER;
1192 v_reg_id NUMBER;
1193 v_retry_interval NUMBER;
1194 v_err_msg VARCHAR2(4000);
1195 BEGIN
1196 INSERT INTO CRM_MPM_JOB_RUN_HISTORY (JOB_NAME, STATUS)
1197 VALUES ('TIMEOUT_JOB', 'RUNNING')
1198 RETURNING RUN_ID INTO v_run_id;
1199 COMMIT;
1200
1201 -- FIX: OPEN/FETCH/CLOSE pattern — no PLS-00364 risk
1202 OPEN c_timeout;
1203 LOOP
1204 FETCH c_timeout
1205 INTO v_log_id, v_reg_id, v_retry_count, v_max_retry, v_retry_interval;
1206 EXIT WHEN c_timeout%NOTFOUND;
1207
1208 UPDATE CRM_MPM_CRM_INTEGRATION_LOG
1209 SET FINAL_STATUS = 'TIMEOUT',
1210 ERROR_CODE = 'TIMEOUT',
1211 ERROR_MESSAGE = 'No callback received within configured timeout window',
1212 IS_FINAL_ATTEMPT = CASE WHEN v_retry_count >= v_max_retry THEN 'Y' ELSE 'N' END,
1213 UPDATED_DATE = SYSTIMESTAMP
1214 WHERE LOG_ID = v_log_id;
1215
1216 IF v_retry_count >= v_max_retry THEN
1217 UPDATE CRM_MPM_CRM_INTEGRATION_LOG
1218 SET FINAL_STATUS = 'EXHAUSTED',
1219 ERROR_CODE = 'EXHAUSTED'
1220 WHERE LOG_ID = v_log_id;
1221 END IF;
1222
1223 v_count := v_count + 1;
1224 END LOOP;
1225 CLOSE c_timeout;
1226
1227 UPDATE CRM_MPM_JOB_RUN_HISTORY
1228 SET END_TIME = SYSTIMESTAMP,
1229 RECORDS_PROCESSED = v_count,
1230 STATUS = 'COMPLETED'
1231 WHERE RUN_ID = v_run_id;
1232
1233 COMMIT;
1234
1235 EXCEPTION
1236 WHEN OTHERS THEN
1237 v_err_msg := SQLERRM;
1238 IF c_timeout%ISOPEN THEN CLOSE c_timeout; END IF;
1239 UPDATE CRM_MPM_JOB_RUN_HISTORY
1240 SET END_TIME = SYSTIMESTAMP,
1241 STATUS = 'ERROR',
1242 ERROR_MESSAGE = SUBSTR(v_err_msg, 1, 4000)
1243 WHERE RUN_ID = v_run_id;
1244 COMMIT;
1245 RAISE;
1246 END RUN_TIMEOUT_JOB;

1249 /* ====================================================================
1250 RUN_RETRY_JOB
1251 ==================================================================== */
1252 PROCEDURE RUN_RETRY_JOB IS
1253 v_run_id NUMBER;
1254 v_processed NUMBER := 0;
1255 v_success NUMBER := 0;
1256 v_failed NUMBER := 0;
1257 v_log_id_new NUMBER;
1258 v_status VARCHAR2(20);
1259 -- FIX: capture SQLERRM
1260 v_err_msg VARCHAR2(4000);
1261 BEGIN
1262 INSERT INTO CRM_MPM_JOB_RUN_HISTORY (JOB_NAME, STATUS)
1263 VALUES ('RETRY_JOB', 'RUNNING')
1264 RETURNING RUN_ID INTO v_run_id;
1265 COMMIT;
1266
1267 FOR rec IN (
1268 SELECT L.LOG_ID, L.REGISTRY_ID, L.SOURCE_RECORD_ID,
1269 L.RETRY_COUNT, L.TRANSACTION_GROUP_ID,
1270 L.ATTEMPT_NO,
1271 R.MAX_RETRY_COUNT, R.RETRY_INTERVAL_MINUTES
1272 FROM CRM_MPM_CRM_INTEGRATION_LOG L
1273 JOIN CRM_MPM_API_REGISTRY R ON R.REGISTRY_ID = L.REGISTRY_ID
1274 WHERE L.FINAL_STATUS IN ('FAILED','TIMEOUT')
1275 AND L.RETRY_COUNT < R.MAX_RETRY_COUNT
1276 AND L.IS_FINAL_ATTEMPT = 'N'
1277 AND L.FINAL_STATUS NOT IN ('RETRY_IN_PROGRESS','RETRY_SENT','EXHAUSTED')
1278 AND (L.NEXT_RETRY_DATE IS NULL OR L.NEXT_RETRY_DATE <= SYSTIMESTAMP)
1279 AND R.IS_ACTIVE = 'Y'
1280 ) LOOP
1281
1282 v_processed := v_processed + 1;
1283
1284 BEGIN
1285 -- Mark original row as final attempt BEFORE calling SEND_TO_APIC
1286 IF rec.RETRY_COUNT + 1 >= rec.MAX_RETRY_COUNT THEN
1287 UPDATE CRM_MPM_CRM_INTEGRATION_LOG
1288 SET IS_FINAL_ATTEMPT = 'Y'
1289 WHERE LOG_ID = rec.LOG_ID;
1290 END IF;
1291
1292 -- FIX: Update NEXT_RETRY_DATE on ORIGINAL row BEFORE calling SEND_TO_APIC
1293 -- This prevents retry job from picking up same row again
1294 -- even if SEND_TO_APIC takes time or the job runs concurrently
1295 UPDATE CRM_MPM_CRM_INTEGRATION_LOG
1296 SET NEXT_RETRY_DATE = SYSTIMESTAMP + (rec.RETRY_INTERVAL_MINUTES / 1440),
1297 RETRY_COUNT = rec.RETRY_COUNT + 1,
1298 IS_FINAL_ATTEMPT = CASE WHEN rec.RETRY_COUNT + 1 >= rec.MAX_RETRY_COUNT
1299 THEN 'Y' ELSE 'N' END,
1300 FINAL_STATUS = 'RETRY_IN_PROGRESS',
1301 UPDATED_DATE = SYSTIMESTAMP
1302 WHERE LOG_ID = rec.LOG_ID;
1303 COMMIT;
1304
1305 -- SEND_TO_APIC creates a NEW log row for this attempt
1306 SEND_TO_APIC(
1307 p_registry_id => rec.REGISTRY_ID,
1308 p_key_value => rec.SOURCE_RECORD_ID,
1309 p_transaction_group_id => rec.TRANSACTION_GROUP_ID,
1310 p_attempt_no => rec.ATTEMPT_NO + 1,
1311 p_log_id_out => v_log_id_new
1312 );
1313
1314 -- FIX: Check status on NEW row (the actual push attempt result)
1315 SELECT FINAL_STATUS INTO v_status
1316 FROM CRM_MPM_CRM_INTEGRATION_LOG
1317 WHERE LOG_ID = v_log_id_new;
1318
1319 IF v_status = 'SENT' THEN
1320 v_success := v_success + 1;
1321 -- Mark original row as RETRY_SENT so it is no longer picked up
1322 UPDATE CRM_MPM_CRM_INTEGRATION_LOG
1323 SET FINAL_STATUS = 'RETRY_SENT',
1324 UPDATED_DATE = SYSTIMESTAMP
1325 WHERE LOG_ID = rec.LOG_ID;
1326 ELSE
1327 v_failed := v_failed + 1;
1328 -- Push failed again — mark original row back to FAILED
1329 -- so it can be retried next interval (if retries remaining)
1330 -- or EXHAUSTED if max reached
1331 IF rec.RETRY_COUNT + 1 >= rec.MAX_RETRY_COUNT THEN
1332 UPDATE CRM_MPM_CRM_INTEGRATION_LOG
1333 SET FINAL_STATUS = 'EXHAUSTED',
1334 ERROR_CODE = 'EXHAUSTED',
1335 IS_FINAL_ATTEMPT = 'Y',
1336 UPDATED_DATE = SYSTIMESTAMP
1337 WHERE LOG_ID = rec.LOG_ID;
1338 ELSE
1339 UPDATE CRM_MPM_CRM_INTEGRATION_LOG
1340 SET FINAL_STATUS = 'FAILED',
1341 UPDATED_DATE = SYSTIMESTAMP
1342 WHERE LOG_ID = rec.LOG_ID;
1343 END IF;
1344 END IF;
1345
1346 COMMIT;
1347
1348 EXCEPTION
1349 WHEN OTHERS THEN
1350 v_failed := v_failed + 1;
1351 NULL;
1352 END;
1353 END LOOP;
1354
1355 UPDATE CRM_MPM_JOB_RUN_HISTORY
1356 SET END_TIME = SYSTIMESTAMP,
1357 RECORDS_PROCESSED = v_processed,
1358 RECORDS_SUCCESS = v_success,
1359 RECORDS_FAILED = v_failed,
1360 STATUS = 'COMPLETED'
1361 WHERE RUN_ID = v_run_id;
1362
1363 COMMIT;
1364
1365 EXCEPTION
1366 WHEN OTHERS THEN
1367 v_err_msg := SQLERRM;
1368 UPDATE CRM_MPM_JOB_RUN_HISTORY
1369 SET END_TIME = SYSTIMESTAMP,
1370 STATUS = 'ERROR',
1371 ERROR_MESSAGE = SUBSTR(v_err_msg, 1, 4000)
1372 WHERE RUN_ID = v_run_id;
1373 COMMIT;
1374 RAISE;
1375 END RUN_RETRY_JOB;
